   83 function estimateTowCapacity(vehicle: string | null | undefined): number | null {
   84   if (!vehicle) return null;
   85   const v = String(vehicle).toLowerCase().replace(/[^a-z0-9]/g, "_");
   86   for (const [key, cap] of Object.entries(TOW_CAPACITY)) {
   87     if (v.includes(key.replace("_", ""))) return cap;
   88   }
   89   return null;

   92 // ─── Step 1: Build SQL filters from the FULL buyer profile ───
   93 function buildListingFilters(profile: Record<string, unknown>): string[] {
   94   const conditions: string[] = [];
   95
   96   // RV Type — hard filter
   97   const allowedTypes = ["travel_trailer","fifth_wheel","class_a","class_b","class_c","toy_hauler","popup_camper
      ","truck_camper"];
   98   if (profile.rvType && allowedTypes.includes(String(profile.rvType))) {
   99     conditions.push(`rv_type = '${String(profile.rvType).replace(/'/g, "''")}'`);
  100   }
  101
  102   // Budget — hard filter, NEVER exceed maxBudget
  103   if (profile.maxBudget && !isNaN(Number(profile.maxBudget))) {
  104     conditions.push(`COALESCE(sale_price, price) <= ${Number(profile.maxBudget)}`);
  105   }
  106   if (profile.minBudget && !isNaN(Number(profile.minBudget))) {
  107     conditions.push(`COALESCE(sale_price, price) >= ${Number(profile.minBudget)}`);
  108   }
  109
  110   // Sleeps — must accommodate the travel party
  111   if (profile.travelers && !isNaN(Number(profile.travelers))) {
  112     conditions.push(`sleeps >= ${Number(profile.travelers)}`);
  113   }
  114
  115   // Length constraints
  116   if (profile.maxLength && !isNaN(Number(profile.maxLength))) {
  117     conditions.push(`length <= ${Number(profile.maxLength)}`);
  118   }
  119   if (profile.minLength && !isNaN(Number(profile.minLength))) {
  120     conditions.push(`length >= ${Number(profile.minLength)}`);
  121   }
  122
  123   // Tow vehicle capacity — prevent recommending RVs they can't tow
  124   const towCap = estimateTowCapacity(profile.towVehicle as string);
  125   if (towCap) {
  126     const towableTypes = ["travel_trailer","fifth_wheel","toy_hauler","popup_camper","truck_camper"];
  127     if (profile.rvType && towableTypes.includes(String(profile.rvType))) {
  128       conditions.push(`COALESCE(gvwr, dry_weight) <= ${towCap}`);
  129     }
  130   }
  131
  132   // New vs used preference (if expressed)
  133   if (profile.condition === "new") {
  134     conditions.push(`condition = 'new'`);
  135   } else if (profile.condition === "used") {
  136     conditions.push(`condition = 'used'`);
  137   }
  138
  139   // Only show available inventory
  140   conditions.push(`inventory_status = 'available'`);
  141
  142   return conditions;
  143 }

  210     model: "claude-sonnet-4-6",
  211     max_tokens: 2048,
  212     system: RERANK_SYSTEM_PROMPT,
  213     messages: [
  214       {

      h} total):\n${JSON.stringify(candidateSummaries, null, 2)}`,
  217       },
  218     ],

  222
  223   try {
  224     const parsed = JSON.parse(rerankText.trim());
  225     const matches = parsed.matches || [];
  226
  227     return matches
  228       .sort((a: { rank: number }, b: { rank: number }) => a.rank - b.rank)

  247 async function getMatchedListings(
  248   profile: Record<string, unknown>
  249 ): Promise<Record<string, unknown>[]> {
  250   const conditions = buildListingFilters(profile);
  251   const whereClause = conditions.length > 0 ? `WHERE ${conditions.join(" AND ")}` : "";
  252
  253   // Pull 15 candidates so the AI has good options to choose from
  254   const recRows = await db.execute(
  255     sql.raw(`SELECT * FROM listings ${whereClause} ORDER BY is_featured DESC, COALESCE(sale_price, price) ASC L
      IMIT 15`)
  256   );
  257
  258   const candidates = (recRows as { rows?: Record<string, unknown>[] }).rows ?? [];
  259
  260   if (candidates.length === 0) return [];
  261
  262   // AI re-ranks and picks the best 3 with personalized descriptions
  263   return rerankWithAI(profile, candidates);

  272     const sessionIdOut = sessionId || `session_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;

  287

  290     const stageMatch = rawText.match(/<stage>([\s\S]*?)<\/stage>/);
  291
  292     let updatedProfile = buyerProfile || {};
  293     if (profileMatch) {
  294       try {
  295         const extracted = JSON.parse(profileMatch[1].trim());
  296         updatedProfile = { ...updatedProfile };
  297         for (const [key, value] of Object.entries(extracted)) {
  298           if (value !== null && value !== undefined) {
  299             (updatedProfile as Record<string, unknown>)[key] = value;
  300           }
  301         }
  302       } catch {
  303         // ignore parse errors
  304       }
  305     }
  306
  307     const stage = (stageMatch ? stageMatch[1].trim() : "want") as string;
  308
  309     // Strip the profile and stage tags from the message
  310     const cleanMessage = rawText
  311       .replace(/<profile>[\s\S]*?<\/profile>/g, "")
  312       .replace(/<stage>[\s\S]*?<\/stage>/g, "")
  313       .trim();
  314
  315     // If we're in the matching stage, fetch recommendations using FULL profile
  316     let recommendations: Record<string, unknown>[] = [];
  317     if ((stage === "matching" || stage === "complete") && updatedProfile) {
  318       recommendations = await getMatchedListings(updatedProfile as Record<string, unknown>);
  319     }
  320
  321     res.json({
  322       message: cleanMessage,
  323       sessionId: sessionIdOut,
  324       updatedProfile,
  325       recommendations,
  326       stage,
  327     });
  328   } catch (err) {
  329     console.error("Outfitter chat error:", err);
